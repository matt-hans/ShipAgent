# OpenAI/Claude Connector Invocation Lifecycle Relay Recovery Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the relay invocation lifecycle, timeout ladder, relay-loss recovery, degraded provider envelopes, desktop relay dispatcher validation, and the single `job_ref` async continuation contract for the OpenAI/Claude connector.

**Architecture:** Plan 2 sits between Plan 1's authenticated relay transport and Plan 7's provider execution/approval flow. Cloud code records every invocation in Redis, dispatches through the Plan 1 `ExecutionTarget` seam, returns schema-valid degraded envelopes for relay failures, and reconciles accepted work after reconnect. Desktop code validates relay envelopes by session, sequence, deadline, duplicate ID, and input hash before dispatching to existing local workflow handlers.

**Tech Stack:** Python 3.12, FastAPI control plane, `redis.asyncio`, Pydantic v2, pytest/pytest-asyncio, canonical registry artifacts generated by `scripts/generate_provider_artifacts.py`.

```python
from __future__ import annotations

from enum import StrEnum
from typing import Any, Literal

from pydantic import BaseModel, ConfigDict, Field


class InvocationState(StrEnum):
    QUEUED = "queued"
    SENT_TO_TARGET = "sent_to_target"
    ACCEPTED = "accepted"
    RUNNING = "running"
    RESULT_RETURNED = "result_returned"
    TARGET_OFFLINE_BEFORE_ACCEPT = "target_offline_before_accept"
    TARGET_DISCONNECTED_MID_CALL = "target_disconnected_mid_call"
    DEADLINE_EXCEEDED = "deadline_exceeded"
    ABANDONED = "abandoned"
    RECOVERED_BY_POLL = "recovered_by_poll"


class ProviderEnvelopeStatus(StrEnum):
    BLOCKED = "blocked"
    UNAVAILABLE = "unavailable"
    PROCESSING = "processing"
    PROCESSING_UNKNOWN = "processing_unknown"


class ProviderErrorReason(StrEnum):
    TARGET_OFFLINE = "target_offline"
    TARGET_DISCONNECTED_MID_CALL = "target_disconnected_mid_call"
    DEADLINE_EXCEEDED = "deadline_exceeded"
    PROCESSING_UNKNOWN = "processing_unknown"
    INVOCATION_ABANDONED = "invocation_abandoned"
    RELAY_ENVELOPE_REJECTED = "relay_envelope_rejected"


TERMINAL_STATES = frozenset(
    {
        InvocationState.RESULT_RETURNED,
        InvocationState.TARGET_OFFLINE_BEFORE_ACCEPT,
        InvocationState.DEADLINE_EXCEEDED,
        InvocationState.ABANDONED,
        InvocationState.RECOVERED_BY_POLL,
    }
)

ALLOWED_TRANSITIONS: dict[InvocationState, frozenset[InvocationState]] = {
    InvocationState.QUEUED: frozenset(
        {
            InvocationState.SENT_TO_TARGET,
            InvocationState.TARGET_OFFLINE_BEFORE_ACCEPT,
            InvocationState.ABANDONED,
        }
    ),
    InvocationState.SENT_TO_TARGET: frozenset(
        {
            InvocationState.ACCEPTED,
            InvocationState.TARGET_OFFLINE_BEFORE_ACCEPT,
            InvocationState.DEADLINE_EXCEEDED,
            InvocationState.ABANDONED,
        }
    ),
    InvocationState.ACCEPTED: frozenset(
        {
            InvocationState.RUNNING,
            InvocationState.RESULT_RETURNED,
            InvocationState.TARGET_DISCONNECTED_MID_CALL,
            InvocationState.DEADLINE_EXCEEDED,
            InvocationState.RECOVERED_BY_POLL,
        }
    ),
    InvocationState.RUNNING: frozenset(
        {
            InvocationState.RESULT_RETURNED,
            InvocationState.TARGET_DISCONNECTED_MID_CALL,
            InvocationState.DEADLINE_EXCEEDED,
            InvocationState.RECOVERED_BY_POLL,
        }
    ),
    InvocationState.TARGET_DISCONNECTED_MID_CALL: frozenset(
        {
            InvocationState.RECOVERED_BY_POLL,
            InvocationState.ABANDONED,
        }
    ),
    InvocationState.DEADLINE_EXCEEDED: frozenset(
        {
            InvocationState.RECOVERED_BY_POLL,
            InvocationState.ABANDONED,
        }
    ),
    InvocationState.RESULT_RETURNED: frozenset(),
    InvocationState.TARGET_OFFLINE_BEFORE_ACCEPT: frozenset(),
    InvocationState.ABANDONED: frozenset(),
    InvocationState.RECOVERED_BY_POLL: frozenset(),
}


class InvocationRecord(BaseModel):
    model_config = ConfigDict(use_enum_values=False, extra="forbid")

    relay_invocation_id: str
    account_id: str
    provider_connection_id: str
    execution_target_id: str
    relay_session_id: str
    tool_name: str
    arguments_hash: str
    idempotency_key: str
    audit_correlation_id: str
    state: InvocationState
    sequence: int = Field(ge=1)
    deadline_epoch_ms: int = Field(gt=0)
    job_ref: str | None = None
    local_job_id: str | None = None
    terminal: bool = False
    last_error_reason: ProviderErrorReason | None = None


ProviderEnvelope = dict[str, str | bool | int]


def transition_record(
    record: InvocationRecord,
    next_state: InvocationState,
    *,
    reason: ProviderErrorReason | None = None,
    local_job_id: str | None = None,
    job_ref: str | None = None,
) -> InvocationRecord:
    allowed = ALLOWED_TRANSITIONS[record.state]
    if next_state not in allowed:
        raise ValueError(
            f"invalid invocation transition: {record.state.value} -> {next_state.value}"
        )
    return record.model_copy(
        update={
            "state": next_state,
            "terminal": next_state in TERMINAL_STATES,
            "last_error_reason": reason,
            "local_job_id": local_job_id if local_job_id is not None else record.local_job_id,
            "job_ref": job_ref if job_ref is not None else record.job_ref,
        }
    )


def build_processing_envelope(*, job_ref: str, poll_after_ms: int) -> ProviderEnvelope:
    return {
        "status": ProviderEnvelopeStatus.PROCESSING.value,
        "job_ref": job_ref,
        "poll_after_ms": poll_after_ms,
    }


def build_degraded_envelope(
    *,
    status: Literal[
        ProviderEnvelopeStatus.BLOCKED,
        ProviderEnvelopeStatus.UNAVAILABLE,
        ProviderEnvelopeStatus.PROCESSING_UNKNOWN,
    ],
    reason: ProviderErrorReason,
    terminal: bool,
    message: str,
    job_ref: str | None = None,
    poll_after_ms: int | None = None,
) -> ProviderEnvelope:
    envelope: ProviderEnvelope = {
        "status": status.value,
        "reason": reason.value,
        "terminal": terminal,
        "message": message,
    }
    if job_ref is not None:
        envelope["job_ref"] = job_ref
    if poll_after_ms is not None:
        envelope["poll_after_ms"] = poll_after_ms
    return envelope
```

```python
    if isinstance(value, bytes):
        value = value.decode("utf-8")

    if not isinstance(decoded, dict):

    return decoded

        if raw is None:

        self,

        job_ref: str | None = None,
```

```python
class JobReferenceRecord(BaseModel):
    model_config = ConfigDict(extra="forbid")

    job_ref: str
    account_id: str
    provider_connection_id: str
    execution_target_id: str
    relay_invocation_id: str
    tool_name: str
    idempotency_key: str

        "processing",
        "processing_unknown",
        "completed",
        "failed",
        "needs_review",
    ] = "processing"


class JobReferenceStore:
    def __init__(self, redis_client, token_urlsafe=secrets.token_urlsafe) -> None:
        self.redis = redis_client
        self._token_urlsafe = token_urlsafe

    async def create(
        self,
        *,
        account_id: str,
        provider_connection_id: str,
        execution_target_id: str,
        relay_invocation_id: str,
        tool_name: str,
        idempotency_key: str,
        local_job_id: str | None,
    ) -> JobReferenceRecord:
        job_ref = f"jobref-{self._token_urlsafe(32)}"
        record = JobReferenceRecord(
            job_ref=job_ref,
            account_id=account_id,
            provider_connection_id=provider_connection_id,
            execution_target_id=execution_target_id,
            relay_invocation_id=relay_invocation_id,
            tool_name=tool_name,
            idempotency_key=idempotency_key,
            local_job_id=local_job_id,
        )
        created = await self.redis.set(
            RedisKey.job_reference(job_ref),
            record.model_dump_json(),
            ex=RedisTtl.PROVIDER_POLL_SECONDS,
            nx=True,
        )
        if not created:
            raise ValueError(f"job reference collision: {job_ref}")
        return record

    async def resolve(
        self,
        job_ref: str,
        *,
        account_id: str,
        provider_connection_id: str,
    ) -> JobReferenceRecord | None:
        raw = await self.redis.get(RedisKey.job_reference(job_ref))
        if raw is None:
            return None
        record = JobReferenceRecord.model_validate(_decode_json(raw))
        if record.account_id != account_id:
            return None
        if record.provider_connection_id != provider_connection_id:
            return None
        return record

    async def get_unscoped(self, job_ref: str) -> JobReferenceRecord | None:
        raw = await self.redis.get(RedisKey.job_reference(job_ref))
        if raw is None:
            return None
        return JobReferenceRecord.model_validate(_decode_json(raw))

    async def save(self, record: JobReferenceRecord) -> JobReferenceRecord:
        await self.redis.set(
            RedisKey.job_reference(record.job_ref),
            record.model_dump_json(),
            ex=RedisTtl.PROVIDER_POLL_SECONDS,
        )
        return record

    async def bind_local_job(
        self,
        job_ref: str,
        *,
        local_job_id: str,
    ) -> JobReferenceRecord:
        record = await self.get_unscoped(job_ref)
        if record is None:
            raise KeyError(f"job reference not found: {job_ref}")
        return await self.save(record.model_copy(update={"local_job_id": local_job_id}))
```

```python
class InvocationLifecycleCoordinator:
    def __init__(
        self,
        *,
        invocation_store: InvocationLifecycleStore,
        job_reference_store: JobReferenceStore,
        timeout_ladder: TimeoutLadder | None = None,
        new_invocation_id: Callable[[], str] = _uuid_text,
        new_idempotency_key: Callable[[], str] = _uuid_text,
        new_audit_correlation_id: Callable[[], str] = _uuid_text,
        now_epoch_ms: Callable[[], int],
    ) -> None:
        self.invocations = invocation_store
        self.job_refs = job_reference_store
        self.timeouts = timeout_ladder or TimeoutLadder()
        self._new_invocation_id = new_invocation_id
        self._new_idempotency_key = new_idempotency_key
        self._new_audit_correlation_id = new_audit_correlation_id
        self._now_epoch_ms = now_epoch_ms

    async def invoke(
        self,
        *,
        target: RelayLifecycleTarget,
        account_id: str,
        provider_connection_id: str,
        tool_name: str,
        arguments: dict[str, Any],
        arguments_hash: str,
        grant_callbacks: GrantCallbacks | None,
        async_contract: bool,
    ) -> dict[str, Any]:

        idempotency_key = self._new_idempotency_key()
        audit_correlation_id = self._new_audit_correlation_id()
        deadline_epoch_ms = self._now_epoch_ms() + int(
            self.timeouts.sync_hard_deadline_seconds * 1000
        )
        record = InvocationRecord(
            relay_invocation_id=relay_invocation_id,
            account_id=account_id,
            provider_connection_id=provider_connection_id,
            execution_target_id=target.execution_target_id,
            relay_session_id=target.relay_session_id,
            tool_name=tool_name,
            arguments_hash=arguments_hash,
            idempotency_key=idempotency_key,
            audit_correlation_id=audit_correlation_id,
            state=InvocationState.QUEUED,
            sequence=1,
            deadline_epoch_ms=deadline_epoch_ms,
        )
        await self.invocations.create(record)
        job_record = await self.job_refs.create(
            account_id=account_id,
            provider_connection_id=provider_connection_id,
            execution_target_id=target.execution_target_id,
            relay_invocation_id=relay_invocation_id,
            tool_name=tool_name,
            idempotency_key=idempotency_key,
            local_job_id=None,
        )
        await self.invocations.transition(
            relay_invocation_id,
            InvocationState.SENT_TO_TARGET,
            job_ref=job_record.job_ref,
        )
        if grant_callbacks is not None:
            await grant_callbacks.reserve(idempotency_key=idempotency_key)

        envelope = RelayInvocationEnvelope(
            relay_session_id=target.relay_session_id,
            sequence=1,
            relay_invocation_id=relay_invocation_id,
            tool_name=tool_name,
            arguments=arguments,
            input_hash=arguments_hash,
            deadline_epoch_ms=deadline_epoch_ms,
            idempotency_key=idempotency_key,
            audit_correlation_id=audit_correlation_id,
            provider_connection_id=provider_connection_id,
            account_id=account_id,
            execution_target_id=target.execution_target_id,
        )
        try:
            ack = await asyncio.wait_for(
                target.dispatch_invocation(envelope),
                timeout=self.timeouts.cloud_send_seconds + self.timeouts.target_accept_seconds,
            )
        except TimeoutError:
            await self.invocations.transition(
                relay_invocation_id,
                InvocationState.TARGET_OFFLINE_BEFORE_ACCEPT,
                reason=ProviderErrorReason.TARGET_OFFLINE,
            )
            if grant_callbacks is not None:
                await grant_callbacks.release(idempotency_key=idempotency_key)
            return build_degraded_envelope(
                status=ProviderEnvelopeStatus.UNAVAILABLE,
                reason=ProviderErrorReason.TARGET_OFFLINE,
                terminal=True,
                message="The bound ShipAgent desktop did not accept the invocation. Ask the user to reopen ShipAgent and retry before the approval expires.",
            )

        if not ack.accepted:
            await self.invocations.transition(
                relay_invocation_id,
                InvocationState.TARGET_OFFLINE_BEFORE_ACCEPT,
                reason=ProviderErrorReason.RELAY_ENVELOPE_REJECTED,
            )
            if grant_callbacks is not None:
                await grant_callbacks.release(idempotency_key=idempotency_key)

                reason=ProviderErrorReason.TARGET_OFFLINE,

            )

            relay_invocation_id,
            InvocationState.ACCEPTED,
            local_job_id=ack.local_job_id,
        )
        if ack.local_job_id is not None:
            await self.job_refs.bind_local_job(
                job_record.job_ref,
                local_job_id=ack.local_job_id,
            )
        if grant_callbacks is not None:
            await grant_callbacks.consume_on_accept(idempotency_key=idempotency_key)

        if async_contract:
            await self.invocations.transition(
                relay_invocation_id,
                InvocationState.RUNNING,
            )
            return build_processing_envelope(
                job_ref=job_record.job_ref,
                poll_after_ms=self.timeouts.poll_after_ms,
            )

        result = await target.await_invocation_result(
            relay_invocation_id,
            timeout_seconds=self.timeouts.sync_hard_deadline_seconds,
        )
        if result is None:
            await self.invocations.transition(
                relay_invocation_id,
                InvocationState.TARGET_DISCONNECTED_MID_CALL,
                reason=ProviderErrorReason.PROCESSING_UNKNOWN,
            )
            current_job_ref = await self.job_refs.get_unscoped(job_record.job_ref)
            if current_job_ref is not None:
                await self.job_refs.save(
                    current_job_ref.model_copy(
                        update={"status": "processing_unknown"}
                    )
                )
            return build_degraded_envelope(
                status=ProviderEnvelopeStatus.PROCESSING_UNKNOWN,
                reason=ProviderErrorReason.PROCESSING_UNKNOWN,
                terminal=False,
                message="ShipAgent accepted the invocation but the relay lost the final response. Poll get_job_status with the job_ref before retrying.",
                job_ref=job_record.job_ref,
                poll_after_ms=self.timeouts.poll_after_ms,
            )

        await self.invocations.transition(
            relay_invocation_id,
            InvocationState.RESULT_RETURNED,
            local_job_id=result.local_job_id,
        )
        if result.local_job_id is not None:
            await self.job_refs.bind_local_job(
                job_record.job_ref,
                local_job_id=result.local_job_id,
            )
        return dict(result.result)
```

```python
class ReconnectReconciler:
    def __init__(
        self,
        invocation_store: InvocationLifecycleStore,
        job_reference_store: JobReferenceStore,
    ) -> None:
        self.invocations = invocation_store
        self.job_refs = job_reference_store

    def can_retry_after_reconnect(self, record: InvocationRecord | None) -> bool:
        if record is None:
            return False
        return record.state is InvocationState.TARGET_OFFLINE_BEFORE_ACCEPT

    async def reconcile(
        self,
        report: RelayReconnectReport,
    ) -> list[InvocationRecord]:
        changed: list[InvocationRecord] = []
        outstanding_by_invocation = {
            job.relay_invocation_id: job for job in report.outstanding_jobs
        }
        for relay_invocation_id in report.accepted_invocation_ids:
            record = await self.invocations.get(relay_invocation_id)
            if record is None:
                continue
            if record.execution_target_id != report.execution_target_id:
                continue
            if record.state not in {
                InvocationState.TARGET_DISCONNECTED_MID_CALL,
                InvocationState.DEADLINE_EXCEEDED,
            }:
                continue
            local_job = outstanding_by_invocation.get(relay_invocation_id)
            updated = await self.invocations.transition(
                relay_invocation_id,
                InvocationState.RECOVERED_BY_POLL,
                reason=ProviderErrorReason.PROCESSING_UNKNOWN,
                local_job_id=local_job.local_job_id if local_job else record.local_job_id,
            )
            if record.job_ref and local_job:
                job_ref_record = await self.job_refs.get_unscoped(record.job_ref)
                if job_ref_record is not None:
                    await self.job_refs.save(
                        job_ref_record.model_copy(
                            update={
                                "local_job_id": local_job.local_job_id,
                                "status": "processing",
                            }
                        )
                    )
            changed.append(updated)
        return changed
```

```python
from __future__ import annotations

import asyncio
from collections.abc import Awaitable, Callable, Mapping
from dataclasses import dataclass
from typing import Any

from src.control_plane.request_controls import hash_arguments
from src.control_plane.relay.protocol import (
    RelayInvocationAck,
    RelayInvocationEnvelope,
    RelayInvocationResult,
    RelayOutstandingJob,
    RelayReconnectReport,
)

LocalRelayHandler = Callable[[dict[str, Any]], Awaitable[dict[str, Any]]]


class RelayEnvelopeRejected(ValueError):
    def __init__(self, reason: str) -> None:
        self.reason = reason
        super().__init__(reason)


@dataclass(frozen=True)
class AcceptedInvocation:
    relay_invocation_id: str
    idempotency_key: str
    local_job_id: str | None = None
    status: str = "running"


class InMemoryAcceptedInvocationStore:
    def __init__(self) -> None:
        self._accepted: dict[str, AcceptedInvocation] = {}

    def has(self, relay_invocation_id: str) -> bool:
        return relay_invocation_id in self._accepted

    def record(self, invocation: AcceptedInvocation) -> None:
        self._accepted[invocation.relay_invocation_id] = invocation

    def all(self) -> list[AcceptedInvocation]:
        return list(self._accepted.values())


class RelayInvocationDispatcher:
    def __init__(
        self,
        *,
        relay_session_id: str,
        execution_target_id: str,
        handlers: Mapping[str, LocalRelayHandler],
        accepted_store: InMemoryAcceptedInvocationStore,
        now_epoch_ms: Callable[[], int],
        outstanding_jobs_provider: Callable[[], list[RelayOutstandingJob]] | None = None,
    ) -> None:
        self.relay_session_id = relay_session_id
        self.execution_target_id = execution_target_id
        self.handlers = dict(handlers)
        self.accepted_store = accepted_store
        self._now_epoch_ms = now_epoch_ms
        self._outstanding_jobs_provider = outstanding_jobs_provider or (lambda: [])
        self._last_sequence = 0
        self._result_tasks: dict[str, asyncio.Task[RelayInvocationResult]] = {}

    async def handle_invocation(
        self,
        envelope: RelayInvocationEnvelope,
    ) -> RelayInvocationAck:
        self._validate(envelope)
        self._last_sequence = envelope.sequence
        self.accepted_store.record(
            AcceptedInvocation(
                relay_invocation_id=envelope.relay_invocation_id,
                idempotency_key=envelope.idempotency_key,
            )
        )
        task = asyncio.create_task(self._run_handler(envelope))
        self._result_tasks[envelope.relay_invocation_id] = task
        return RelayInvocationAck(
            relay_invocation_id=envelope.relay_invocation_id,
            accepted=True,
            local_job_id=None,
            reason=None,
        )

    async def await_result(
        self,
        relay_invocation_id: str,
        *,
        timeout_seconds: float,
    ) -> RelayInvocationResult | None:
        task = self._result_tasks.get(relay_invocation_id)
        if task is None:
            return None
        try:
            return await asyncio.wait_for(asyncio.shield(task), timeout=timeout_seconds)
        except TimeoutError:
            return None

    def build_reconnect_report(self, *, relay_session_id: str) -> RelayReconnectReport:
        return RelayReconnectReport(
            relay_session_id=relay_session_id,
            execution_target_id=self.execution_target_id,
            accepted_invocation_ids=[
                invocation.relay_invocation_id
                for invocation in self.accepted_store.all()
            ],
            outstanding_jobs=self._outstanding_jobs_provider(),
        )

    def _validate(self, envelope: RelayInvocationEnvelope) -> None:
        if envelope.relay_session_id != self.relay_session_id:
            raise RelayEnvelopeRejected("wrong_session")
        if envelope.execution_target_id != self.execution_target_id:
            raise RelayEnvelopeRejected("wrong_execution_target")
        if self.accepted_store.has(envelope.relay_invocation_id):
            raise RelayEnvelopeRejected("duplicate_invocation")
        if envelope.sequence <= self._last_sequence:
            raise RelayEnvelopeRejected("non_increasing_sequence")
        if envelope.deadline_epoch_ms <= self._now_epoch_ms():
            raise RelayEnvelopeRejected("deadline_expired")
        if envelope.input_hash != hash_arguments(envelope.arguments):
            raise RelayEnvelopeRejected("input_hash_mismatch")
        if envelope.tool_name not in self.handlers:
            raise RelayEnvelopeRejected("unknown_tool")

    async def _run_handler(self, envelope: RelayInvocationEnvelope) -> RelayInvocationResult:
        handler = self.handlers[envelope.tool_name]
        result = await handler(envelope.arguments)
        return RelayInvocationResult(
            relay_invocation_id=envelope.relay_invocation_id,
            result=result,
            local_job_id=result.get("job_id") if isinstance(result.get("job_id"), str) else None,
            error_reason=None,
        )
```
