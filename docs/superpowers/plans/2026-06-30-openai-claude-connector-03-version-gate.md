# Relay Version Gate Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Enforce relay heartbeat version compatibility before provider tool dispatch and return a schema-valid terminal `target_update_required` result when the active execution target cannot satisfy a tool contract.

**Architecture:** The canonical registry remains the source of truth: target-required `ToolContract.minimum_capabilities` entries define each tool's compatibility matrix and generated provider artifacts publish that metadata. A pure `src/control_plane/relay/version_gate.py` evaluates Plan 1 heartbeat metadata, and the hosted MCP boundary calls it before request controls or handlers so incompatible targets never receive tool inputs. The failure path returns a provider-safe envelope that validates against each target-required tool's output schema.

**Tech Stack:** Python 3.12, Pydantic v2, jsonschema, FastMCP, pytest/pytest-asyncio, Redis-backed Plan 1 relay session store.

```python
    @field_validator("minimum_capabilities")
    @classmethod
    def _minimum_capabilities_must_be_stable(
        cls, value: dict[str, str]
    ) -> dict[str, str]:
        normalized: dict[str, str] = {}
        for key, required_version in value.items():
            if not _MINIMUM_CAPABILITY_KEY_RE.fullmatch(key):
                raise ValueError(
                    "minimum_capabilities keys must be lowercase dotted identifiers"
                )
            if key == "ups_boundary_contract":
                if not _UPS_BOUNDARY_CONTRACT_RE.fullmatch(required_version):
                    raise ValueError(
                        "minimum_capabilities values for ups_boundary_contract "
                        "must look like hosted-v1"
                    )
            elif not _SEMVER_RE.fullmatch(required_version):
                raise ValueError(
                    "minimum_capabilities values must be semantic versions"
                )
            normalized[key] = required_version
        return dict(sorted(normalized.items()))
```

```python
"""Relay heartbeat compatibility gate for provider-callable tools."""

from __future__ import annotations

import re
from collections.abc import Iterable, Mapping
from dataclasses import dataclass
from typing import Protocol

from src.registry.models import ToolContract

VERSION_METADATA_REQUIREMENTS = frozenset(
    {"shipagent_core", "registry_contract", "ups_boundary_contract"}
)

_SEMVER_RE = re.compile(r"^(0|[1-9]\d*)\.(0|[1-9]\d*)\.(0|[1-9]\d*)$")


@dataclass(frozen=True)
class CapabilityMismatch:
    name: str
    required: str
    actual: str


@dataclass(frozen=True)
class CompatibilityDecision:
    compatible: bool
    required: dict[str, str]
    missing: tuple[str, ...] = ()
    outdated: tuple[CapabilityMismatch, ...] = ()


@dataclass(frozen=True)
class TargetVersionMetadata:
    shipagent_core_version: str
    registry_contract_version: str
    ups_boundary_contract_version: str
    capability_versions: dict[str, str]

    @classmethod
    def from_heartbeat(
        cls,
        *,
        shipagent_core_version: str,
        registry_contract_version: str,
        ups_boundary_contract_version: str,
        capabilities: Mapping[str, str] | Iterable[str],
    ) -> "TargetVersionMetadata":
        return cls(
            shipagent_core_version=shipagent_core_version,
            registry_contract_version=registry_contract_version,
            ups_boundary_contract_version=ups_boundary_contract_version,
            capability_versions=normalize_capability_versions(capabilities),
        )


class TargetMetadataProvider(Protocol):
    async def get_active_target_version_metadata(
        self, account_id: str
    ) -> TargetVersionMetadata | None:
        """Return active target heartbeat metadata for an account."""


class VersionGate:
    def __init__(self, *, metadata_provider: TargetMetadataProvider) -> None:
        self._metadata_provider = metadata_provider

    async def check(
        self,
        *,
        account_id: str,
        contract: ToolContract,
    ) -> dict[str, object] | None:
        if not contract.execution_target_required and not contract.minimum_capabilities:
            return None

        target = await self._metadata_provider.get_active_target_version_metadata(
            account_id
        )
        if target is None:
            return None

        decision = evaluate_target_compatibility(contract=contract, target=target)
        if decision.compatible:
            return None
        return target_update_required_envelope(decision)


def normalize_capability_versions(
    capabilities: Mapping[str, str] | Iterable[str],
) -> dict[str, str]:
    if isinstance(capabilities, Mapping):
        return {
            str(name): str(version)
            for name, version in sorted(capabilities.items())
        }

    normalized: dict[str, str] = {}
    for item in capabilities:
        name, separator, version = str(item).partition("@")
        normalized[name] = version if separator else "0.0.0"
    return dict(sorted(normalized.items()))


def evaluate_target_compatibility(
    *,
    contract: ToolContract,
    target: TargetVersionMetadata,
) -> CompatibilityDecision:
    required = dict(contract.minimum_capabilities)
    missing: list[str] = []
    outdated: list[CapabilityMismatch] = []

    for name, required_version in required.items():
        actual_version = _actual_version_for_requirement(name, target)
        if actual_version is None:
            missing.append(name)
            continue
        if not _satisfies_requirement(name, actual_version, required_version):
            outdated.append(
                CapabilityMismatch(
                    name=name,
                    required=required_version,
                    actual=actual_version,
                )
            )

    return CompatibilityDecision(
        compatible=not missing and not outdated,
        required=required,
        missing=tuple(sorted(missing)),
        outdated=tuple(sorted(outdated, key=lambda item: item.name)),
    )


def target_update_required_envelope(
    decision: CompatibilityDecision,
) -> dict[str, object]:
    required_capabilities = sorted(
        {
            *decision.missing,
            *(mismatch.name for mismatch in decision.outdated),
        }
    )
    return {
        "status": "unavailable",
        "reason": "target_update_required",
        "terminal": True,
        "message": (
            "Your ShipAgent runtime must be updated before this tool can run. "
            "Update ShipAgent Desktop, reconnect Cloud AI Features, then ask again."
        ),
        "executionTarget": {"state": "update_required"},
        "required_capabilities": required_capabilities,
    }


def _actual_version_for_requirement(
    name: str,
    target: TargetVersionMetadata,
) -> str | None:
    if name == "shipagent_core":
        return target.shipagent_core_version
    if name == "registry_contract":
        return target.registry_contract_version
    if name == "ups_boundary_contract":
        return target.ups_boundary_contract_version
    return target.capability_versions.get(name)


def _satisfies_requirement(name: str, actual: str, required: str) -> bool:
    if name == "ups_boundary_contract":
        return actual == required

    actual_semver = _parse_semver(actual)
    required_semver = _parse_semver(required)
    if actual_semver is None or required_semver is None:
        return False
    return actual_semver >= required_semver


def _parse_semver(version: str) -> tuple[int, int, int] | None:
    match = _SEMVER_RE.fullmatch(version)
    if match is None:
        return None
    return tuple(int(part) for part in match.groups())
```
